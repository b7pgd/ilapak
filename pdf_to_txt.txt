    // Utility Format Size
    function formatBytes(bytes) {
      if (bytes === 0) return '0 Bytes';
      const k = 1024;
      const sizes = ['Bytes', 'KB', 'MB', 'GB'];
      const i = Math.floor(Math.log(bytes) / Math.log(k));
      return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
    }

    /* =========================================================
       3. PDF PARSING & SPATIAL RECONSTRUCTION LOGIC
       ========================================================= */
    async function processPDF(file) {
      if (typeof pdfjsLib === 'undefined') {
        throw new Error('Library PDF.js gagal dimuat dari CDN.');
      }

      updateProgress(5, 'Membaca file PDF...');
      const arrayBuffer = await file.arrayBuffer();
      const loadingTask = pdfjsLib.getDocument({ data: arrayBuffer });
      const pdf = await loadingTask.promise;
      
      const numPages = pdf.numPages;
      const pdfJsonData = {
        source: {
          filename: file.name,
          type: 'pdf',
          pages: numPages
        },
        pages: []
      };

      let shouldOcr = enableOcrCheckbox.checked;

      for (let i = 1; i <= numPages; i++) {
        updateProgress(Math.round((i / numPages) * 80), `Memproses halaman ${i} dari ${numPages}...`);
        const page = await pdf.getPage(i);
        const textContent = await page.getTextContent();
        
        let rawItems = textContent.items.map(item => ({
          text: item.str,
          x: item.transform[4],
          y: item.transform[5],
          width: item.width || 0,
          height: item.height || Math.sqrt(item.transform[0]*item.transform[0] + item.transform[1]*item.transform[1]) || 10
        }));

        // Cek jika halaman adalah scanned PDF (text layer kosong)
        const combinedRawText = rawItems.map(it => it.text).join('').trim();
        if (combinedRawText.length < 10 && shouldOcr) {
          updateProgress(Math.round((i / numPages) * 80), `Menjalankan OCR pada halaman ${i}...`);
          rawItems = await runOcrOnPage(page);
        }

        const reconstructedText = reconstructPdfPageText(rawItems);

        pdfJsonData.pages.push({
          page: i,
          text: reconstructedText,
          items: rawItems
        });
      }

      // Deteksi & Filter Header / Footer berulang jika diaktifkan
      if (removeHeadersFootersCheckbox.checked && pdfJsonData.pages.length > 1) {
        updateProgress(90, 'Membersihkan header/footer berulang...');
        filterRepeatedHeadersFooters(pdfJsonData);
      }

      updateProgress(95, 'Menyusun hasil teks akhir...');
      extractedJsonObject = pdfJsonData;
      jsonOutput.value = JSON.stringify(extractedJsonObject, null, 2);

      // JSON -> TXT Conversion
      finalTxtString = convertPdfJsonToTxt(extractedJsonObject);
      txtOutput.value = finalTxtString;
      updateProgress(100, 'Selesai!');
    }

    // Rekonstruksi Teks PDF berdasarkan Koordinat X dan Y
    function reconstructPdfPageText(items) {
      if (!items || items.length === 0) return '';

      // Clean empty strings
      const validItems = items.filter(it => it.text && it.text.trim().length > 0);
      if (validItems.length === 0) return '';

      // Tentukan Layout Multi-kolom dengan memeriksa distribusi X
      // Grouping sederhana berdasarkan X jika ada gap horizontal besar
      // Pertama, kelompokkan baris vertikal (Y-tolerance)
      const yTolerance = 4; // Beda Y <= 4px dianggap baris sama
      
      // Urutkan dari atas ke bawah (Y besar ke kecil di PDF.js)
      const sortedByY = [...validItems].sort((a, b) => b.y - a.y);

      const lines = [];
      sortedByY.forEach(item => {
        let matchedLine = lines.find(line => Math.abs(line.y - item.y) <= yTolerance);
        if (matchedLine) {
          matchedLine.items.push(item);
        } else {
          lines.push({
            y: item.y,
            items: [item]
          });
        }
      });

      // Urutkan line dari atas ke bawah
      lines.sort((a, b) => b.y - a.y);

      let reconstructedPage = '';
      let prevLineY = null;
      let avgLineHeight = 12;

      lines.forEach((line, index) => {
        // Urutkan item dalam satu baris dari kiri ke kanan (X kecil ke besar)
        line.items.sort((a, b) => a.x - b.x);

        let lineStr = '';
        let prevItemRight = null;

        line.items.forEach(item => {
          if (prevItemRight !== null) {
            const gap = item.x - prevItemRight;
            // Jika gap horizontal melebihi 2px, tambahkan spasi
            if (gap > 2) {
              lineStr += ' ';
            }
          }
          lineStr += item.text;
          prevItemRight = item.x + (item.width || (item.text.length * 5));
        });

        // Deteksi Spasi Paragraf vs Line Break Biasa
        if (prevLineY !== null) {
          const verticalGap = prevLineY - line.y;
          if (verticalGap > avgLineHeight * 1.6) {
            reconstructedPage += '\n\n'; // Paragraph break
          } else {
            reconstructedPage += '\n'; // Line break
          }
        }

        reconstructedPage += lineStr.trim();
        prevLineY = line.y;
      });

      return reconstructedPage;
    }

    // Filter Repeated Headers and Footers
    function filterRepeatedHeadersFooters(pdfJson) {
      const pageCount = pdfJson.pages.length;
      const headerCandidates = {};
      const footerCandidates = {};

      pdfJson.pages.forEach(p => {
        const lines = p.text.split('\n').map(l => l.trim()).filter(l => l.length > 0);
        if (lines.length > 0) {
          const top = lines[0];
          const bottom = lines[lines.length - 1];
          headerCandidates[top] = (headerCandidates[top] || 0) + 1;
          footerCandidates[bottom] = (footerCandidates[bottom] || 0) + 1;
        }
      });

      // Jika muncul lebih dari 60% jumlah halaman, bersihkan
      const threshold = pageCount * 0.6;
      
      pdfJson.pages.forEach(p => {
        let lines = p.text.split('\n');
        if (lines.length > 0 && headerCandidates[lines[0].trim()] >= threshold) {
          lines.shift();
        }
        if (lines.length > 0 && footerCandidates[lines[lines.length - 1].trim()] >= threshold) {
          lines.pop();
        }
        p.text = lines.join('\n').trim();
      });
    }

    // OCR Fallback via Tesseract.js
    async function runOcrOnPage(pdfPage) {
      if (typeof Tesseract === 'undefined') {
        throw new Error('Tesseract.js tidak dimuat. Harap periksa koneksi internet/CDN.');
      }

      const viewport = pdfPage.getViewport({ scale: 1.5 });
      const canvas = document.createElement('canvas');
      const context = canvas.getContext('2d');
      canvas.height = viewport.height;
      canvas.width = viewport.width;

      await pdfPage.render({ canvasContext: context, viewport: viewport }).promise;

      const worker = await Tesseract.createWorker('eng');
      const ret = await worker.recognize(canvas);
      await worker.terminate();

      const ocrItems = [];
      if (ret.data && ret.data.words) {
        ret.data.words.forEach(w => {
          ocrItems.push({
            text: w.text,
            x: w.bbox.x0,
            y: viewport.height - w.bbox.y0, // Invert Y agar sesuai koordinat PDF.js
            width: w.bbox.x1 - w.bbox.x0,
            height: w.bbox.y1 - w.bbox.y0
          });
        });
      }
      return ocrItems;
    }

    /* =========================================================
       4. DOCX PARSING LOGIC
       ========================================================= */
    async function processDOCX(file) {
      if (typeof mammoth === 'undefined') {
        throw new Error('Library Mammoth.js gagal dimuat dari CDN.');
      }

      updateProgress(20, 'Membaca file DOCX...');
      const arrayBuffer = await file.arrayBuffer();

      updateProgress(50, 'Mengekstrak struktur dokumen DOCX...');
      
      // Gunakan convertToHtml dari mammoth untuk mendapatkan tag struktur (p, h1, li)
      const result = await mammoth.convertToHtml({ arrayBuffer: arrayBuffer });
      const htmlString = result.value;

      const docxJsonData = {
        source: {
          filename: file.name,
          type: 'docx'
        },
        blocks: []
      };

      // Parse HTML string ke DOM Parser untuk ekstrak struktur JSON
      const parser = new DOMParser();
      const doc = parser.parseFromString(htmlString, 'text/html');
      
      doc.body.childNodes.forEach(node => {
        if (node.nodeType === Node.ELEMENT_NODE) {
          const tagName = node.tagName.toLowerCase();
          const textContent = node.textContent.trim();

          if (!textContent) return;

          if (/^h[1-6]$/.test(tagName)) {
            docxJsonData.blocks.push({
              type: 'heading',
              level: parseInt(tagName.charAt(1)),
              text: textContent
            });
          } else if (tagName === 'ul' || tagName === 'ol') {
            node.querySelectorAll('li').forEach(li => {
              docxJsonData.blocks.push({
                type: 'list',
                text: li.textContent.trim()
              });
            });
          } else {
            docxJsonData.blocks.push({
              type: 'paragraph',
              text: textContent
            });
          }
        }
      });

      updateProgress(80, 'Menyusun teks bersih...');
      extractedJsonObject = docxJsonData;
      jsonOutput.value = JSON.stringify(extractedJsonObject, null, 2);

      finalTxtString = convertDocxJsonToTxt(docxJsonData);
      txtOutput.value = finalTxtString;
      updateProgress(100, 'Selesai!');
    }

    /* =========================================================
       5. JSON -> TXT CONVERTER (CLEAN & NORMALIZED)
       ========================================================= */
    function convertPdfJsonToTxt(jsonObject) {
      if (!jsonObject.pages || !Array.isArray(jsonObject.pages)) return '';
      
      return jsonObject.pages
        .map(page => page.text)
        .filter(text => text && text.trim().length > 0)
        .join('\n\n')
        .replace(/[ \t]+/g, ' ') // Normalisasi spasi berlebihan pada baris
        .replace(/\n{3,}/g, '\n\n') // Maksimal 2 newline berturut-turut
        .trim();
    }

    function convertDocxJsonToTxt(jsonObject) {
      if (!jsonObject.blocks || !Array.isArray(jsonObject.blocks)) return '';

      const lines = jsonObject.blocks.map(block => {
        if (block.type === 'heading') {
          return `${block.text.toUpperCase()}`;
        } else if (block.type === 'list') {
          return `• ${block.text}`;
        } else {
          return block.text;
        }
      });

      return lines
        .join('\n\n')
        .replace(/[ \t]+/g, ' ')
        .replace(/\n{3,}/g, '\n\n')
        .trim();
    }

    // Generic JSON -> TXT Dispatcher
    function convertJsonToTxt(jsonObject) {
      if (jsonObject.source && jsonObject.source.type === 'pdf') {
        return convertPdfJsonToTxt(jsonObject);
      } else if (jsonObject.source && jsonObject.source.type === 'docx') {
        return convertDocxJsonToTxt(jsonObject);
      } else if (jsonObject.pages) {
        return convertPdfJsonToTxt(jsonObject);
      } else if (jsonObject.blocks) {
        return convertDocxJsonToTxt(jsonObject);
      } else {
        throw new Error('Format JSON tidak dikenali. Harus memiliki properti "pages" atau "blocks".');
      }
    }

    /* =========================================================
       9. MANUAL MODE: JSON -> TXT
       ========================================================= */
    btnConvertManualJson.addEventListener('click', () => {
      hideError();
      const rawJson = jsonManualInput.value.trim();
      if (!rawJson) {
        showError('JSON Input manual masih kosong.');
        return;
      }

      try {
        const parsed = JSON.parse(rawJson);
        const txt = convertJsonToTxt(parsed);
        txtOutput.value = txt;
        extractedJsonObject = parsed;
        jsonOutput.value = JSON.stringify(parsed, null, 2);
      } catch (err) {
        showError('JSON tidak valid: ' + err.message);
      }
    });

    /* =========================================================
       13 & 14. COPY & DOWNLOAD HANDLERS
       ========================================================= */
    async function copyToClipboard(text, btnElement) {
      const originalText = btnElement.innerText;
      try {
        if (navigator.clipboard && navigator.clipboard.writeText) {
          await navigator.clipboard.writeText(text);
        } else {
          // Fallback kompatibilitas
          const tempArea = document.createElement('textarea');
          tempArea.value = text;
          document.body.appendChild(tempArea);
          tempArea.select();
          document.execCommand('copy');
          document.body.removeChild(tempArea);
        }
        btnElement.innerText = 'Copied!';
        setTimeout(() => {
          btnElement.innerText = originalText;
        }, 2000);
      } catch (err) {
        alert('Gagal menyalin teks ke clipboard.');
      }
    }

    function downloadFile(content, fileName, contentType) {
      const blob = new Blob([content], { type: contentType });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = fileName;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    }

    function getBaseFileName() {
      if (selectedFile) {
        return selectedFile.name.substring(0, selectedFile.name.lastIndexOf('.'));
      }
      return 'converted-document';
    }

    btnCopyJson.addEventListener('click', () => {
      if (!jsonOutput.value) return;
      copyToClipboard(jsonOutput.value, btnCopyJson);
    });

    btnDownloadJson.addEventListener('click', () => {
      if (!jsonOutput.value) return;
      downloadFile(jsonOutput.value, `${getBaseFileName()}.json`, 'application/json');
    });

    btnCopyTxt.addEventListener('click', () => {
      if (!txtOutput.value) return;
      copyToClipboard(txtOutput.value, btnCopyTxt);
    });

    btnDownloadTxt.addEventListener('click', () => {
      if (!txtOutput.value) return;
      downloadFile(txtOutput.value, `${getBaseFileName()}.txt`, 'text/plain');
    });
